  @Test
  public void reloadXmlOmitHeaderNonHosting() throws Exception {
    String coll = "rel_reload_xml_omitheader";
    CollectionAdminRequest.createCollection(coll, CONFIG, 1, 1)
        .setCreateNodeSet(cluster.getJettySolrRunner(0).getNodeName())
        .process(cluster.getSolrClient());
    cluster.waitForActiveCollection(coll, 1, 1);
    try {
      JettySolrRunner hosting = cluster.getJettySolrRunner(0);
      JettySolrRunner nonHosting = cluster.getJettySolrRunner(1);
      Replica replica = getCollectionState(coll).getSlice("shard1").getReplicas().iterator().next();
      assertEquals(hosting.getNodeName(), replica.getNodeName());
      assertFalse(localCoreNames(nonHosting).contains(replica.getCoreName()));
      SolrCore before = hosting.getCoreContainer().getCore(replica.getCoreName());
      try {
        assertNotNull(before);
        String url = nonHosting.getBaseURLV2() + "/collections/" + coll + "/reload?wt=xml&omitHeader=true";
        ContentResponse r = post(url, "{}", "*/*");
        log(
            "XML_RELOAD_OMITHEADER target="
                + nonHosting.getNodeName()
                + " httpStatus="
                + r.getStatus()
                + " contentType="
                + String.valueOf(r.getHeaders().get("Content-Type"))
                + " url="
                + url);
        log("XML_RELOAD_OMITHEADER body=" + prefix(r.getContentAsString()));
        assertEquals(200, r.getStatus());
        Document doc = parseXml(r.getContentAsString());
        assertEquals("response", doc.getDocumentElement().getNodeName());
        Double headers =
            (Double)
                XPathFactory.newInstance()
                    .newXPath()
                    .evaluate(
                        "count(/response/lst[@name='responseHeader'])",
                        doc,
                        XPathConstants.NUMBER);
        assertEquals("responseHeader must be omitted", 0.0, headers, 0.0);
        try (SolrCore after = hosting.getCoreContainer().getCore(replica.getCoreName())) {
          assertNotNull(after);
          assertNotSame("reload must replace the hosted core", before, after);
        }
      } finally {
        if (before != null) {
          before.close();
        }
      }
    } finally {
      CollectionAdminRequest.deleteCollection(coll).process(cluster.getSolrClient());
    }
  }

  private void runCase(Op op, Topo topo, Wt wt, Mode mode) throws Exception {
    String caseId =
        op.name().toLowerCase()
            + "_"
            + topo.name().toLowerCase()
            + "_"
            + wt.name().toLowerCase()
            + "_"
            + mode.name().toLowerCase();
    String coll = "rel_" + caseId;
    String backupName = caseId;
    String restored = coll + "_restored";
    String asyncId = mode == Mode.ASYNC ? "r17241_" + caseId : null;
    JettySolrRunner a = cluster.getJettySolrRunner(0);
    JettySolrRunner b = cluster.getJettySolrRunner(1);
    JettySolrRunner c = cluster.getJettySolrRunner(2);
    JettySolrRunner target = topo == Topo.HOSTING ? a : b;
    String path =
        op == Op.BACKUP
            ? "/collections/" + coll + "/backups/" + backupName + "/versions"
            : "/collections/" + coll + "/shards/shard1/replicas";
    Map<String, Object> body = new LinkedHashMap<>();
    if (op == Op.BACKUP) {
      body.put("repository", REPO);
      body.put("location", backupLocation);
    } else {
      body.put("node", c.getNodeName());
    }
    if (asyncId != null) {
      body.put("async", asyncId);
    }
    String json = Utils.toJSONString(body);
    log(
        "CASE "
            + caseId
            + " op="
            + op
            + " topo="
            + topo
            + " wt="
            + wt
            + " mode="
            + mode
            + " collection="
            + coll
            + " target="
            + target.getNodeName()
            + " pinnedNode="
            + c.getNodeName()
            + " path="
            + path
            + " body="
            + json);

    CollectionAdminRequest.createCollection(coll, CONFIG, 1, 1)
        .setCreateNodeSet(a.getNodeName())
        .process(cluster.getSolrClient());
    cluster.waitForActiveCollection(coll, 1, 1);
    indexDocs(coll);

    Replica original = getCollectionState(coll).getSlice("shard1").getReplicas().iterator().next();
    assertEquals(a.getNodeName(), original.getNodeName());
    boolean clusterStateHosts =
        allReplicas(getCollectionState(coll)).stream()
            .anyMatch(r -> r.getNodeName().equals(target.getNodeName()));
    boolean localCore = localCoreNames(target).stream().anyMatch(n -> n.startsWith(coll + "_shard"));
    log(
        "TOPOLOGY "
            + caseId
            + " target="
            + target.getNodeName()
            + " clusterStateHosts="
            + clusterStateHosts
            + " localCorePresent="
            + localCore
            + " expectedHosting="
            + (topo == Topo.HOSTING));
    assertEquals(topo == Topo.HOSTING, clusterStateHosts);
    assertEquals(topo == Topo.HOSTING, localCore);
    int replicasBefore = allReplicas(getCollectionState(coll)).size();

    Result result = null;
    Exception failure = null;
    try {
      result = send(wt, target, path, json, asyncId);
      log(
          "RESPONSE "
              + caseId
              + " transport="
              + result.transport()
              + " url="
              + result.url()
              + " contentType="
              + result.contentType()
              + " responseStatus="
              + result.responseStatus()
              + " asyncIdPresent="
              + result.asyncIdPresent()
              + " body="
              + result.bodyPrefix());
    } catch (Exception e) {
      failure = e;
      log("RESPONSE " + caseId + " EXCEPTION " + e);
    }

    String terminal = null;
    if (asyncId != null) {
      terminal = awaitTerminal(asyncId);
      log("ASYNC_STATUS " + caseId + " id=" + asyncId + " state=" + terminal);
    }

    boolean effect;
    if (op == Op.ADDREPLICA) {
      effect = verifyAddReplica(coll, a, c, replicasBefore);
    } else {
      effect = verifyBackup(coll, backupName, restored);
    }
    log("EFFECT " + caseId + " operationCompleted=" + effect);

    CollectionAdminRequest.deleteCollection(coll).process(cluster.getSolrClient());
    if (failure != null) {
      throw failure;
    }
    assertEquals(0, (int) result.responseStatus());
    if (asyncId != null) {
      assertTrue("async id must be in the response", result.asyncIdPresent());
      assertEquals(RequestStatusState.COMPLETED.getKey(), terminal);
    }
    assertTrue("operation must complete", effect);
  }

  private Result send(Wt wt, JettySolrRunner target, String path, String json, String asyncId)
      throws Exception {
    switch (wt) {
      case JAVABIN:
        return sendJavabin(target, path, json, asyncId);
      case XML:
        return sendXml(target, path, json, asyncId);
      default:
        return sendJson(target, path, json, asyncId);
    }
  }

  private Result sendJavabin(JettySolrRunner target, String path, String json, String asyncId)
      throws Exception {
    try (HttpJettySolrClient client = new HttpJettySolrClient.Builder(target.getBaseUrl().toString()).build()) {
      V2Request request =
          new V2Request.Builder(path)
              .withMethod(SolrRequest.METHOD.POST)
              .withParams(params("wt", "javabin"))
              .withPayload(json)
              .build();
      request.setResponseParser(new JavaBinResponseParser());
      V2Response response = request.process(client);
      NamedList<Object> body = response.getResponse();
      Object status = ((NamedList<?>) body.get("responseHeader")).get("status");
      return new Result(
          "javabin(SolrJ JavaBinResponseParser, base=" + target.getBaseUrl() + ")",
          target.getBaseUrl() + "/api" + path + "?wt=javabin",
          "application/vnd.apache.solr.javabin (validated by SolrJ parser)",
          status == null ? null : Integer.valueOf(status.toString()),
          asyncId != null && body.toString().contains(asyncId),
          prefix(body.toString()));
    }
  }

  private Result sendXml(JettySolrRunner target, String path, String json, String asyncId)
      throws Exception {
    String url = target.getBaseURLV2() + path + "?wt=xml";
    ContentResponse r = post(url, json, "*/*");
    Document doc = parseXml(r.getContentAsString());
    String status =
        (String)
            XPathFactory.newInstance()
                .newXPath()
                .evaluate("/response/lst[@name='responseHeader']/int[@name='status']", doc, XPathConstants.STRING);
    return new Result(
        "xml(JDK HttpClient, Accept */*)",
        url,
        String.valueOf(r.getHeaders().get("Content-Type")),
        r.getStatus() == 200 && !status.isEmpty() ? Integer.valueOf(status) : null,
        asyncId != null && r.getContentAsString().contains(asyncId),
        prefix(r.getContentAsString()));
  }

  private Result sendJson(JettySolrRunner target, String path, String json, String asyncId)
      throws Exception {
    String url = target.getBaseURLV2() + path;
    ContentResponse r = post(url, json, null);
    Object parsed = Utils.fromJSONString(r.getContentAsString());
    Object status = Utils.getObjectByPath(parsed, true, "/responseHeader/status");
    return new Result(
        "json(JDK HttpClient, no wt)",
        url,
        String.valueOf(r.getHeaders().get("Content-Type")),
        status == null ? null : Integer.valueOf(status.toString()),
        asyncId != null && r.getContentAsString().contains(asyncId),
        prefix(r.getContentAsString()));
  }

  private ContentResponse post(String url, String body, String accept) throws Exception {
    return cluster
        .getJettySolrRunner(0)
        .getSolrClient()
        .getHttpClient()
        .newRequest(url)
        .method(HttpMethod.POST)
        .headers(h -> {
          if (accept != null) {
            h.add("Accept", accept);
          }
        })
        .body(new StringRequestContent("application/json", body))
        .send();
  }

  private String awaitTerminal(String asyncId) throws Exception {
    RequestStatusState state = null;
    for (int i = 0; i < 240; i++) {
      state =
          CollectionAdminRequest.requestStatus(asyncId)
              .process(cluster.getSolrClient())
              .getRequestStatus();
      if (state == RequestStatusState.COMPLETED || state == RequestStatusState.FAILED) {
        break;
      }
      Thread.sleep(500);
    }
    return state == null ? "null" : state.getKey();
  }

  private boolean verifyAddReplica(String coll, JettySolrRunner a, JettySolrRunner c, int before)
      throws Exception {
    for (int i = 0; i < 240; i++) {
      DocCollection dc = getCollectionState(coll);
      List<Replica> replicas = new ArrayList<>(allReplicas(dc));
      if (replicas.size() == before + 1) {
        for (Replica r : replicas) {
          if (!r.getNodeName().equals(a.getNodeName())
              && r.getNodeName().equals(c.getNodeName())
              && r.getState() == Replica.State.ACTIVE) {
            return true;
          }
        }
      }
      Thread.sleep(500);
    }
    return false;
  }

  private boolean verifyBackup(String coll, String backupName, String restored) {
    try {
      CollectionAdminRequest.restoreCollection(restored, backupName)
          .setLocation(backupLocation)
          .setRepositoryName(REPO)
          .setConfigName(CONFIG)
          .process(cluster.getSolrClient());
      cluster.waitForActiveCollection(restored, 1, 1);
      Set<String> ids = new TreeSet<>();
      QueryResponse qr =
          cluster.getSolrClient().query(restored, new SolrQuery("*:*").setRows(100));
      for (SolrDocument d : qr.getResults()) {
        ids.add(String.valueOf(d.getFieldValue("id")));
      }
      log("RESTORE " + backupName + " numFound=" + qr.getResults().getNumFound() + " ids=" + ids);
      CollectionAdminRequest.deleteCollection(restored).process(cluster.getSolrClient());
      return qr.getResults().getNumFound() == DOC_COUNT && ids.size() == DOC_COUNT;
    } catch (Exception e) {
      log("RESTORE " + backupName + " FAILED " + e);
      return false;
    }
  }

  private void indexDocs(String coll) throws Exception {
    List<SolrInputDocument> docs = new ArrayList<>();
    for (int i = 0; i < DOC_COUNT; i++) {
      SolrInputDocument d = new SolrInputDocument();
      d.addField("id", "doc" + i);
      d.addField("text", "value " + i);
      docs.add(d);
    }
    cluster.getSolrClient().add(coll, docs);
    cluster.getSolrClient().commit(coll);
  }

  private static List<Replica> allReplicas(DocCollection dc) {
    List<Replica> out = new ArrayList<>();
    for (var slice : dc.getSlices()) {
      out.addAll(slice.getReplicas());
    }
    return out;
  }

  private static Set<String> localCoreNames(JettySolrRunner runner) {
    return new TreeSet<>(runner.getCoreContainer().getAllCoreNames());
  }

  private static Document parseXml(String body) throws Exception {
    DocumentBuilderFactory f = DocumentBuilderFactory.newInstance();
    f.setNamespaceAware(false);
    return f.newDocumentBuilder().parse(new InputSource(new StringReader(body)));
  }

  private static String prefix(String s) {
    String flat = s.replace('\n', ' ');
    return flat.length() > 300 ? flat.substring(0, 300) + "..." : flat;
  }
